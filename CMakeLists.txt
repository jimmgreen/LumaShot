function(verify_ocr_asset name expected)
    file(SHA256 "${OCR_DEPS}/${name}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "OCR dependency checksum mismatch: ${name}. Run scripts/setup-ocr.ps1 to repair.")
    endif()
endfunction()

# Default to the verified operator-reduced 1.22 runtime, without altering models.
# An explicit cache override can still select the official vendor runtime.
set(LUMASHOT_REDUCED_OCR "${CMAKE_CURRENT_SOURCE_DIR}/deps/ocr-runtime-reduced/onnxruntime.dll")
set(LUMASHOT_OCR_RUNTIME "${LUMASHOT_REDUCED_OCR}" CACHE FILEPATH "Validated ONNX Runtime 1.22 DLL")
if(LUMASHOT_OCR_RUNTIME STREQUAL LUMASHOT_REDUCED_OCR)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/deps/ocr-runtime-reduced/manifest.json" runtime_manifest)
    file(SHA256 "${LUMASHOT_OCR_RUNTIME}" runtime_sha256)
    string(JSON expected_runtime_sha256 GET "${runtime_manifest}" sha256)
    if(NOT runtime_sha256 STREQUAL expected_runtime_sha256)
        message(FATAL_ERROR "Reduced OCR runtime hash differs from its verified manifest")
    endif()
    foreach(model det.onnx rec.onnx)
        file(SHA256 "${OCR_DEPS}/${model}" model_sha256)
        string(JSON expected_model_sha256 GET "${runtime_manifest}" models "${model}")
        if(NOT model_sha256 STREQUAL expected_model_sha256)
            message(FATAL_ERROR "Reduced OCR runtime model mismatch: ${model}")
        endif()
    endforeach()
endif()
if(NOT EXISTS "${LUMASHOT_OCR_RUNTIME}")
    message(FATAL_ERROR "OCR runtime does not exist: ${LUMASHOT_OCR_RUNTIME}")
